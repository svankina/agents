# Agent Unification Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One source of truth in `~/src/agents` for the global instruction files and skills of Claude Code, Codex, Pi, and omp, installed by a generator (`agents-sync`) with drift detection, plus symlink-based per-project convention.

**Architecture:** A stdlib-only Python tool (`bin/agents_sync.py`) composes each agent's global file as banner + `shared/AGENTS.md` + `<agent>/local.md`, preserving foreign managed blocks (e.g. the fetcher plugin's), and maintains per-skill symlinks from `shared/skills/` into each agent's skill dir plus `~/AGENTS.md`/`~/CLAUDE.md` home symlinks. Content from the three existing instruction files is merged into `shared/` + per-agent `local.md`.

**Tech Stack:** Python 3.12 stdlib, pytest, git. No third-party deps.

**Spec:** `docs/superpowers/specs/2026-07-03-agent-unification-design.md`

## Global Constraints

- **No sudo anywhere in this plan.** Nothing here needs privileges.
- Never delete a real (non-symlink) directory or file automatically; conflicts are reported for manual resolution. Manual deletions use `trash-put` (recoverable), never `rm -rf`.
- `agents-sync` is the only writer of the four generated targets; it must never write inside foreign `<!-- name:begin -->…<!-- name:end -->` blocks.
- Python: stdlib only. Script must run as `python3 bin/agents_sync.py` and via the `bin/agents-sync` wrapper.
- All paths configurable via `--repo` / `--home` so tests never touch the real `$HOME`.
- Generated banner first line starts exactly with: `<!-- GENERATED by agents-sync`
- Exit codes: `0` ok/clean, `1` check found problems (or runtime error), `2` sync blocked by drift/conflict without `--force`.
- Work happens in worktree `~/src/agents/.worktrees/agent-unification` (branch `agent-unification` off `shared-report-server`) until Task 7 merges it back; the first real `sync` run (Task 7) must run from the canonical checkout `~/src/agents`, because skill symlinks embed that path.

```python
import re
from dataclasses import dataclass
from pathlib import Path

AGENTS = ["claude", "codex", "pi", "omp"]

BANNER_PREFIX = "<!-- GENERATED by agents-sync"

FOREIGN_BLOCK_RE = re.compile(
    r"<!--\s*([A-Za-z0-9_-]+):begin\b[^>]*-->\n?.*?<!--\s*\1:end\s*-->\n?",
    re.DOTALL,
)


@dataclass
class Target:
    agent: str
    path: Path
    local_md: Path


@dataclass
class Config:
    repo: Path
    home: Path

    @property
    def shared_md(self) -> Path:
        return self.repo / "shared" / "AGENTS.md"

    @property
    def shared_skills(self) -> Path:
        return self.repo / "shared" / "skills"

    def targets(self) -> list[Target]:
        paths = {
            "claude": self.home / ".claude" / "CLAUDE.md",
            "codex": self.home / ".codex" / "AGENTS.md",
            "pi": self.home / ".config" / "pi" / "AGENTS.md",
            "omp": self.home / ".omp" / "agent" / "AGENTS.md",
        }
        return [Target(a, paths[a], self.repo / a / "local.md") for a in AGENTS]

    def skill_dests(self) -> list[Path]:
        return [
            self.home / ".claude" / "skills",
            self.home / ".codex" / "skills",
            self.home / ".omp" / "agent" / "skills",
            self.repo / "pi" / "skills",
        ]

    def home_links(self) -> list[tuple[Path, Path]]:
        return [
            (self.home / "AGENTS.md", self.repo / "home" / "AGENTS.md"),
            (self.home / "CLAUDE.md", Path("AGENTS.md")),  # relative link
        ]


def banner(agent: str) -> str:
    return (
        f"<!-- GENERATED by agents-sync — do not edit here. Source: "
        f"~/src/agents/shared/AGENTS.md + ~/src/agents/{agent}/local.md. "
        f"Edit those, then run agents-sync. -->\n"
    )


def extract_foreign_blocks(text: str) -> list[str]:
    return [m.group(0).rstrip("\n") + "\n" for m in FOREIGN_BLOCK_RE.finditer(text)]


def compose(agent: str, shared_text: str, local_text: str,
            foreign_blocks: list[str]) -> str:
    parts = [banner(agent), "\n", shared_text.rstrip("\n") + "\n"]
    if local_text.strip():
        parts += ["\n", local_text.rstrip("\n") + "\n"]
    for block in foreign_blocks:
        parts += ["\n", block]
    return "".join(parts)


def expected_content(cfg: Config, t: Target) -> str:
    shared = cfg.shared_md.read_text()
    local = t.local_md.read_text() if t.local_md.exists() else ""
    current = t.path.read_text() if t.path.exists() else ""
    return compose(t.agent, shared, local, extract_foreign_blocks(current))


def target_state(t: Target, expected: str) -> str:
    if not t.path.exists():
        return "missing"
    current = t.path.read_text()
    if current == expected:
        return "clean"
    if current.startswith(BANNER_PREFIX):
        return "drifted"
    return "foreign"
```

```python
import os


def desired_skill_links(cfg: Config) -> list[tuple[Path, Path]]:
    links = []
    if cfg.shared_skills.is_dir():
        for skill in sorted(p for p in cfg.shared_skills.iterdir() if p.is_dir()):
            for dest in cfg.skill_dests():
                links.append((dest / skill.name, skill))
    return links


def link_state(link: Path, target: Path, repo: Path) -> str:
    if not link.is_symlink():
        if link.exists():
            return "conflict"
        return "missing"
    raw = os.readlink(link)
    try:
        resolved = link.resolve(strict=True)
    except OSError:
        resolved = None
    want = target if target.is_absolute() else (link.parent / target)
    if resolved is not None and resolved == want.resolve():
        return "clean"
    if raw.startswith(str(repo)):
        return "stale"
    return "conflict"


def apply_link(link: Path, target: Path) -> None:
    link.parent.mkdir(parents=True, exist_ok=True)
    if link.is_symlink():
        link.unlink()
    os.symlink(target, link)


def stale_repo_links(cfg: Config) -> list[Path]:
    valid = {link for link, _ in desired_skill_links(cfg)}
    out = []
    for dest in cfg.skill_dests():
        if not dest.is_dir():
            continue
        for entry in sorted(dest.iterdir()):
            if entry in valid or not entry.is_symlink():
                continue
            raw = os.readlink(entry)
            if raw.startswith(str(cfg.shared_skills)) and not entry.exists():
                out.append(entry)
    return out
```

```python
            bad += 1
    for orphan in stale_repo_links(cfg):
        print(f"[orphan] {orphan}")
        bad += 1
    if bad == 0:
        print("clean")
    return 1 if bad else 0


def cmd_diff(cfg: Config) -> int:
    for t, exp, state in _iter_target_items(cfg):
        if state == "clean":
            continue
        current = t.path.read_text() if t.path.exists() else ""
        sys.stdout.writelines(difflib.unified_diff(
            current.splitlines(keepends=True), exp.splitlines(keepends=True),
            fromfile=str(t.path), tofile=f"generated:{t.agent}"))
    return 0


def cmd_sync(cfg: Config, force: bool) -> int:
    blocked = 0
    for t, exp, state in _iter_target_items(cfg):
        if state == "clean":
            continue
        if state in ("drifted", "foreign") and not force:
            print(f"[blocked:{state}] {t.path} — re-run with --force after "
                  f"reviewing (agents-sync diff)")
            blocked += 1
            continue
        t.path.parent.mkdir(parents=True, exist_ok=True)
        t.path.write_text(exp)
        print(f"[wrote] {t.path}")
    for link, target, state in _iter_link_items(cfg):
        if state == "clean":
            continue
        if state == "conflict":
            print(f"[conflict] {link} exists and is not ours — resolve "
                  f"manually (trash-put it), then re-run")
            blocked += 1
            continue
        if state == "stale" and not force:
            print(f"[blocked:stale] {link} — re-run with --force")
            blocked += 1
            continue
        apply_link(link, target)
        print(f"[linked] {link} -> {target}")
    for orphan in stale_repo_links(cfg):
        orphan.unlink()
        print(f"[pruned] {orphan}")
    return 2 if blocked else 0


def main(argv: list[str]) -> int:
    p = argparse.ArgumentParser(prog="agents-sync", description=__doc__)
    p.add_argument("command", choices=["sync", "check", "diff"])
    p.add_argument("--force", action="store_true")
    p.add_argument("--repo", type=Path, default=Path.home() / "src" / "agents")
    p.add_argument("--home", type=Path, default=Path.home())
    args = p.parse_args(argv)
    cfg = Config(repo=args.repo.resolve(), home=args.home.resolve())
    if args.command == "check":
        return cmd_check(cfg)
    if args.command == "diff":
        return cmd_diff(cfg)
    return cmd_sync(cfg, force=args.force)


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```

Note: `test_check_flags_drift_and_sync_blocks_without_force` expects a drifted target to be rewritten by `sync --force` — the hand edit is *outside* a foreign block, so it is intentionally discarded; `expected_content` re-harvests only `name:begin/end` blocks.
